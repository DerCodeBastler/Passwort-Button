cmake_minimum_required(VERSION 3.17)
if (NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
  set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
endif()
if (NOT PICO_SDK_PATH)
  set(PICO_SDK_PATH ${CMAKE_CURRENT_LIST_DIR}/../pico-sdk)
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(keepass_pico C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
if (NOT PICO_BOARD STREQUAL "pico2" OR NOT PICO_PLATFORM STREQUAL "rp2350-arm-s")
  message(FATAL_ERROR "Dieses Paket ist ausschließlich für Pico 2 / RP2350 ARM vorgesehen.")
endif()
add_executable(keepass_pico main.c usb_descriptors.c)
target_include_directories(keepass_pico PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(keepass_pico pico_stdlib hardware_flash hardware_sync pico_unique_id tinyusb_device)
target_compile_options(keepass_pico PRIVATE -Wall -Wextra -Werror)
# Kein UART, kein USB-stdio, kein Reset-Interface und kein 1200-Baud-Bootloaderwechsel.
pico_enable_stdio_usb(keepass_pico 0)
pico_enable_stdio_uart(keepass_pico 0)
pico_set_program_name(keepass_pico "KeePass-Taster Pico 2")
pico_set_program_version(keepass_pico "1.1.0")
pico_add_extra_outputs(keepass_pico)
set_target_properties(keepass_pico PROPERTIES LINKER_LANGUAGE C)
